cmake_minimum_required(VERSION 3.25)
project(tower_of_the_sorcerer VERSION 0.9 LANGUAGES C CXX)
set(TSW_BUILD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../build")
set(FETCHCONTENT_BASE_DIR "${CMAKE_CURRENT_BINARY_DIR}/_deps" CACHE PATH "Dependency build directory")
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_CURRENT_BINARY_DIR)
    message(FATAL_ERROR "Use a preset or a build directory under ../build/cpp")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TSW_GUI "Build the raylib desktop game" ON)
option(TSW_SANITIZERS "Enable address and undefined behavior checks" OFF)
add_library(tsw_core src/game.cpp src/combat.cpp src/items.cpp src/contact.cpp src/mouse_walk.cpp src/hero_orb.cpp
    src/floor_checks.cpp src/scenes.cpp src/npcs.cpp src/save.cpp src/original_save.cpp src/save_slots.cpp
    src/game_state.cpp src/rankings.cpp src/localization.cpp src/achievements.cpp)
target_include_directories(tsw_core PUBLIC include PRIVATE src)
if(MSVC)
    target_compile_options(tsw_core PRIVATE /W4 /utf-8)
    if(TSW_SANITIZERS)
        string(REGEX REPLACE "/RTC[1su]+" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
        target_compile_options(tsw_core PUBLIC /fsanitize=address /Zi)
        target_link_options(tsw_core PUBLIC /INCREMENTAL:NO /DEBUG)
    endif()
else()
    target_compile_options(tsw_core PRIVATE -Wall -Wextra -Wpedantic)
    if(TSW_SANITIZERS)
        target_compile_options(tsw_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(tsw_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()
add_executable(tsw_probe src/probe.cpp)
target_link_libraries(tsw_probe PRIVATE tsw_core)
include(FetchContent)
find_package(glaze 7.2.2 EXACT CONFIG QUIET)
if(NOT glaze_FOUND)
    FetchContent_Declare(glaze URL https://github.com/stephenberry/glaze/archive/refs/tags/v7.2.2.tar.gz
        URL_HASH SHA512=b247761321f6569b8f5141cb3ae8eebd73a7d46996013b1dc43d4b9585b8f5f5ed0e280af245c6e8149a548eb808c8387cfd857394d98e127f96d09d8adf3047
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(glaze)
endif()
target_link_libraries(tsw_core PRIVATE glaze::glaze)
add_library(tsw_replay src/replay.cpp src/replay_player.cpp src/recorder.cpp src/settings.cpp)
target_link_libraries(tsw_replay PUBLIC tsw_core PRIVATE glaze::glaze)
option(TSW_PLANNER "Build offline native route planning tools" OFF)
target_link_libraries(tsw_probe PRIVATE tsw_replay)
add_library(tsw_mcp src/mcp.cpp src/mcp_observation.cpp src/mcp_runtime.cpp)
target_link_libraries(tsw_mcp PUBLIC tsw_replay PRIVATE glaze::glaze)
add_executable(tsw_cli src/cli.cpp)
target_link_libraries(tsw_cli PRIVATE tsw_mcp)
if(MINGW)
    target_link_options(tsw_cli PRIVATE -static-libgcc -static-libstdc++)
endif()
foreach(target IN ITEMS tsw_replay tsw_mcp tsw_cli)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    # MSVC's instrumented STL must agree across the test runner and our objects.
    # Build Catch2 with ASan instead of mixing it with the ordinary vcpkg library.
    if(NOT (MSVC AND TSW_SANITIZERS))
        find_package(Catch2 3 CONFIG QUIET)
    endif()
    if(NOT Catch2_FOUND)
        FetchContent_Declare(Catch2 URL https://github.com/catchorg/Catch2/archive/refs/tags/v3.13.0.tar.gz
            URL_HASH SHA512=7eea385d79d88a5690cde131fe7ccda97d5c54ea09d6f515000d7bf07c828809d61c1ac99912c1ee507cf933f61c1c47ecdcc45df7850ffa82714034b0fccf35
            DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
        FetchContent_MakeAvailable(Catch2)
        list(APPEND CMAKE_MODULE_PATH "${catch2_SOURCE_DIR}/extras")
        if(MSVC AND TSW_SANITIZERS)
            foreach(catch_target IN ITEMS Catch2 Catch2WithMain)
                target_compile_options(${catch_target} PRIVATE /fsanitize=address /Zi)
            endforeach()
        endif()
    endif()
    add_executable(tsw_tests tests/core_tests.cpp tests/mouse_walk_tests.cpp tests/hero_orb_tests.cpp tests/tile_variant_tests.cpp tests/replay_tests.cpp tests/mcp_tests.cpp
        tests/ranking_tests.cpp tests/ranking_game_tests.cpp tests/ranking_replay_tests.cpp tests/localization_tests.cpp tests/achievement_tests.cpp)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-temp")
    target_compile_definitions(tsw_tests PRIVATE TSW_TEST_TEMP="${CMAKE_CURRENT_BINARY_DIR}/test-temp")
    target_link_libraries(tsw_tests PRIVATE tsw_mcp Catch2::Catch2WithMain glaze::glaze)
    if(MSVC)
        target_compile_options(tsw_tests PRIVATE /utf-8)
    endif()
    include(Catch)
    catch_discover_tests(tsw_tests TEST_PREFIX "core: " PROPERTIES LABELS "core")
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cpp_localization_catalogs COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/tools/compile_locales.py" --check)
        set_tests_properties(cpp_localization_catalogs PROPERTIES LABELS "localization" TIMEOUT 30)
        add_test(NAME comparison_boundary_guard COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/tools/comparison_boundary.py" --self-test)
        set_tests_properties(comparison_boundary_guard PROPERTIES LABELS "fidelity" TIMEOUT 30)
        set(replay_cases "${CMAKE_CURRENT_SOURCE_DIR}/../tests/replays")
        add_test(NAME replay_door COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_replay.py"
            "${replay_cases}/first-floor-door.jsonl" "${replay_cases}/first-floor-door.expected.json"
            --cli $<TARGET_FILE:tsw_cli>)
        add_test(NAME replay_blocked_prefix COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_replay.py"
            "${replay_cases}/first-floor-door.jsonl" "${replay_cases}/first-floor-door-blocked.expected.json"
            --cli $<TARGET_FILE:tsw_cli> --limit 3)
        set_tests_properties(replay_door replay_blocked_prefix PROPERTIES LABELS "replay" TIMEOUT 60)
        add_test(NAME mcp_stdio COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_mcp.py"
            --cli $<TARGET_FILE:tsw_cli>)
        set_tests_properties(mcp_stdio PROPERTIES LABELS "mcp" TIMEOUT 60)
        add_test(NAME mcp_reload COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_mcp_reload.py"
            --cli $<TARGET_FILE:tsw_cli>)
        set_tests_properties(mcp_reload PROPERTIES LABELS "mcp" TIMEOUT 120)
    endif()
endif()
if(TSW_PLANNER)
    add_subdirectory(planner)
endif()
if(TSW_GUI)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(raylib_USE_STATIC_LIBS ON)
    find_package(raylib 5.5 CONFIG QUIET)
    if(NOT raylib_FOUND)
      FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz
        URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
      FetchContent_MakeAvailable(raylib)
      set(RAYLIB_LICENSE "${raylib_SOURCE_DIR}/LICENSE")
    else()
      find_package(glfw3 CONFIG REQUIRED)
      if(VCPKG_TARGET_TRIPLET MATCHES "windows-static$")
        # raylib 5.5's package has one unqualified path and repeats it in its
        # link interface. Select the matching CRT library for each VS config.
        set(raylib_root "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
        find_library(TSW_RAYLIB_RELEASE NAMES raylib PATHS "${raylib_root}/lib" NO_DEFAULT_PATH REQUIRED)
        find_library(TSW_RAYLIB_DEBUG NAMES raylib PATHS "${raylib_root}/debug/lib" NO_DEFAULT_PATH REQUIRED)
        set_target_properties(raylib PROPERTIES
          IMPORTED_CONFIGURATIONS "DEBUG;RELEASE"
          IMPORTED_LOCATION_DEBUG "${TSW_RAYLIB_DEBUG}"
          IMPORTED_IMPLIB_DEBUG "${TSW_RAYLIB_DEBUG}"
          IMPORTED_LOCATION_RELEASE "${TSW_RAYLIB_RELEASE}"
          IMPORTED_IMPLIB_RELEASE "${TSW_RAYLIB_RELEASE}"
          MAP_IMPORTED_CONFIG_RELWITHDEBINFO RELEASE
          MAP_IMPORTED_CONFIG_MINSIZEREL RELEASE
          INTERFACE_LINK_LIBRARIES "")
      endif()
      find_file(RAYLIB_LICENSE NAMES copyright
        PATHS "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/raylib" REQUIRED NO_DEFAULT_PATH)
    endif()
    FetchContent_Declare(tinysoundfont URL https://github.com/schellingb/TinySoundFont/archive/853a0a171759f1ddba0de1442133a75912bbeffa.tar.gz
        URL_HASH SHA256=76c356df524f71f2a34f67fa0cb2cb946036fe7a65d2930a849569c5e742fa57
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(tinysoundfont)
    find_path(TSW_MINIAUDIO_INCLUDE_DIR NAMES miniaudio.h REQUIRED)
    find_file(TSW_MINIAUDIO_LICENSE NAMES copyright
        PATHS "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/miniaudio" REQUIRED NO_DEFAULT_PATH)
    include(cmake/Soundfont.cmake)
    include(cmake/EmbeddedMedia.cmake)
    if(raylib_FOUND)
        find_path(TSW_STB_INCLUDE_DIR NAMES stb_image.h REQUIRED)
    else()
        set(TSW_STB_INCLUDE_DIR "${raylib_SOURCE_DIR}/src/external")
    endif()
    add_library(tsw_media src/assets.cpp src/embedded_media.cpp)
    target_include_directories(tsw_media PUBLIC src)
    target_include_directories(tsw_media SYSTEM PRIVATE "${TSW_STB_INCLUDE_DIR}")
    target_link_libraries(tsw_media PUBLIC raylib)
    if(raylib_FOUND)
        target_link_libraries(tsw_media PUBLIC glfw)
        if(WIN32)
            target_link_libraries(tsw_media PUBLIC winmm)
        endif()
    endif()
    add_library(tsw_audio src/audio.cpp src/audio_mixer.cpp)
    target_include_directories(tsw_audio SYSTEM PRIVATE ${tinysoundfont_SOURCE_DIR} ${TSW_MINIAUDIO_INCLUDE_DIR})
    target_link_libraries(tsw_audio PUBLIC tsw_core tsw_media)
    add_library(tsw_desktop src/presentation.cpp src/desktop_session.cpp)
    target_link_libraries(tsw_desktop PUBLIC tsw_audio tsw_replay)
    add_executable(tsw_game src/desktop.cpp)
    target_link_libraries(tsw_game PRIVATE tsw_desktop)
    tsw_embed_media(tsw_game)
    foreach(target IN ITEMS tsw_media tsw_audio tsw_desktop tsw_game)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /utf-8)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endforeach()
    if(MSVC AND TSW_SANITIZERS)
        target_compile_options(tsw_media PRIVATE /fsanitize=address /Zi)
    endif()
    if(WIN32)
        set(presentation_assets "${original_directory}")
    else()
        set(presentation_assets "${TSW_BUILD_ROOT}/assets")
    endif()
    add_custom_command(TARGET tsw_game POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${TSW_MINIAUDIO_LICENSE} $<TARGET_FILE_DIR:tsw_game>/miniaudio-LICENSE.txt
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${SOUNDFONT_ROOT}/LICENSE.txt $<TARGET_FILE_DIR:tsw_game>/GeneralUser-GS-LICENSE.txt
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${tinysoundfont_SOURCE_DIR}/LICENSE $<TARGET_FILE_DIR:tsw_game>/TinySoundFont-LICENSE.txt
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${RAYLIB_LICENSE} $<TARGET_FILE_DIR:tsw_game>/raylib-LICENSE.txt)
    if(MINGW)
        # The optional test target does not exist when BUILD_TESTING is disabled.
        target_link_options(tsw_game PRIVATE -static-libgcc -static-libstdc++)
        target_link_options(tsw_probe PRIVATE -static-libgcc -static-libstdc++)
        if(BUILD_TESTING)
            target_link_options(tsw_tests PRIVATE -static-libgcc -static-libstdc++)
        endif()
    endif()
    if(BUILD_TESTING)
        add_executable(tsw_audio_tests tests/audio_tests.cpp)
        target_link_libraries(tsw_audio_tests PRIVATE tsw_audio Catch2::Catch2)
        target_include_directories(tsw_audio_tests SYSTEM PRIVATE
            ${tinysoundfont_SOURCE_DIR} ${TSW_MINIAUDIO_INCLUDE_DIR})
        target_compile_definitions(tsw_audio_tests PRIVATE TSW_AUDIO_OUTPUT="${CMAKE_CURRENT_BINARY_DIR}/audio-checks")
        tsw_embed_media(tsw_audio_tests)
        catch_discover_tests(tsw_audio_tests TEST_PREFIX "audio: " TEST_SPEC "~[device]"
            PROPERTIES LABELS "audio" TIMEOUT 300)
        add_executable(tsw_verify tests/verify.cpp tests/presentation_checks.cpp tests/localization_checks.cpp
            tests/asset_checks.cpp tests/desktop_smoke.cpp)
        target_link_libraries(tsw_verify PRIVATE tsw_desktop)
        target_include_directories(tsw_verify SYSTEM PRIVATE ${TSW_MINIAUDIO_INCLUDE_DIR})
        target_compile_definitions(tsw_verify PRIVATE TSW_CHECK_OUTPUT="${CMAKE_CURRENT_BINARY_DIR}/desktop-checks")
        tsw_embed_media(tsw_verify)
        foreach(target IN ITEMS tsw_audio_tests tsw_verify)
            if(MSVC)
                target_compile_options(${target} PRIVATE /W4 /utf-8)
                target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
            else()
                target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
            endif()
            if(MINGW)
                target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
            endif()
        endforeach()
        add_test(NAME original_presentation COMMAND tsw_verify --assets "${presentation_assets}"
            --presentation "${CMAKE_CURRENT_BINARY_DIR}/presentation-checks")
        set_tests_properties(original_presentation PROPERTIES TIMEOUT 60 LABELS "desktop")
        add_test(NAME localized_presentation COMMAND tsw_verify --assets "${presentation_assets}"
            --localization "${CMAKE_CURRENT_BINARY_DIR}/localization-checks")
        set_tests_properties(localized_presentation PROPERTIES TIMEOUT 120 LABELS "desktop;localization")
        if(WIN32 AND EXISTS "${TSW_BUILD_ROOT}/assets/original-atlas.png")
            add_test(NAME original_assets COMMAND tsw_verify --assets "${original_directory}"
                --compare-assets "${TSW_BUILD_ROOT}/assets"
                --asset-check "${CMAKE_CURRENT_BINARY_DIR}/asset-checks")
            add_test(NAME extracted_presentation COMMAND tsw_verify --assets "${TSW_BUILD_ROOT}/assets"
                --presentation "${CMAKE_CURRENT_BINARY_DIR}/extracted-presentation-checks")
            set_tests_properties(original_assets extracted_presentation PROPERTIES TIMEOUT 60 LABELS "desktop")
        endif()
        add_test(NAME audio_device COMMAND tsw_audio_tests "[device]")
        set_tests_properties(audio_device PROPERTIES TIMEOUT 30 LABELS "audio")
        add_test(NAME window_resize COMMAND tsw_verify --assets "${presentation_assets}"
            --window "${CMAKE_CURRENT_BINARY_DIR}/window-checks")
        set_tests_properties(window_resize PROPERTIES TIMEOUT 60 LABELS "desktop")
        add_test(NAME desktop_scenarios COMMAND tsw_verify --assets "${presentation_assets}"
            --smoke all --output "${CMAKE_CURRENT_BINARY_DIR}/desktop-checks")
        # DesktopSession opens an audio device, so -LE audio must exclude this suite.
        set_tests_properties(desktop_scenarios PROPERTIES TIMEOUT 90 LABELS "desktop;audio")
    endif()
endif()
